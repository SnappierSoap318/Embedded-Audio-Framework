cmake_minimum_required(VERSION 3.20)
project(eaf VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(EAF_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(EAF_THREAD_SANITIZE "Thread sanitizer (separate build)" OFF)
option(EAF_ENABLE_LMS "Build SlimProto protocol support" ON)
option(EAF_ENABLE_BT_INGRESS "Build bounded SBC packet ingress" ON)
option(EAF_ENABLE_SENDPIN "Build Sendspin player protocol support" ON)
if(EAF_SANITIZE AND EAF_THREAD_SANITIZE)
  message(FATAL_ERROR "Select only one sanitizer mode")
endif()
find_package(Threads REQUIRED)
include(CheckSymbolExists)
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
check_symbol_exists(sem_clockwait "semaphore.h" EAF_HAS_SEM_CLOCKWAIT)
unset(CMAKE_REQUIRED_DEFINITIONS)
unset(CMAKE_REQUIRED_LIBRARIES)
if(NOT EAF_HAS_SEM_CLOCKWAIT)
  message(FATAL_ERROR "Linux HAL requires sem_clockwait for monotonic semaphore deadlines")
endif()
add_compile_options(-Wall -Wextra -Werror -Wconversion -Wstrict-prototypes -Wshadow -Wdouble-promotion)
if(EAF_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(EAF_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(eaf STATIC core/eaf_types.c core/eaf_reservoir.c core/eaf_graph.c core/eaf_control.c
  core/eaf_dsp.c hal/common/hal_atomic_c11.c hal/linux/hal_os_linux.c
  hal/linux/hal_file_linux.c hal/common/sink_null.c hal/linux/hal_tcp_linux.c
  apps/decoders/dec_wav.c apps/player/eaf_player.c)
target_include_directories(eaf PUBLIC include)
target_link_libraries(eaf PUBLIC Threads::Threads m)
if(EAF_ENABLE_LMS)
  target_sources(eaf PRIVATE apps/lms/lms_protocol.c apps/lms/lms_client.c)
  add_executable(eaf_lms_play platform/native_linux/play_lms.c)
  target_link_libraries(eaf_lms_play PRIVATE eaf)
endif()
if(EAF_ENABLE_BT_INGRESS)
  target_sources(eaf PRIVATE apps/bt/bt_ingress.c apps/bt/bt_decoder.c)
endif()
if(EAF_ENABLE_SENDPIN)
  target_sources(eaf PRIVATE apps/sendspin/sendspin_sync.c apps/sendspin/sendspin_protocol.c
    apps/sendspin/sendspin_ws.c apps/sendspin/sendspin_client.c apps/sendspin/sendspin_player.c)
endif()
option(EAF_ENABLE_ALSA "Build optional Linux ALSA output when available" ON)
if(EAF_ENABLE_ALSA)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(ALSA QUIET IMPORTED_TARGET alsa)
  endif()
  if(ALSA_FOUND)
    target_sources(eaf PRIVATE hal/linux/sink_alsa.c)
    target_link_libraries(eaf PUBLIC PkgConfig::ALSA)
  else()
    message(STATUS "ALSA development files unavailable; building null output only")
  endif()
endif()
set(EAF_NATIVE_BACKEND platform/native_linux/native_output_null.c)
if(EAF_ENABLE_ALSA AND ALSA_FOUND)
  set(EAF_NATIVE_BACKEND platform/native_linux/native_output_alsa.c)
endif()
add_library(eaf_native_output STATIC platform/native_linux/native_output.c ${EAF_NATIVE_BACKEND})
target_link_libraries(eaf_native_output PUBLIC eaf)
if(TARGET eaf_lms_play)
  target_link_libraries(eaf_lms_play PRIVATE eaf_native_output)
endif()
include(cmake/libsbc.cmake)
add_executable(eaf_demo platform/native_linux/board_config.c)
target_link_libraries(eaf_demo PRIVATE eaf)
add_executable(eaf_play platform/native_linux/play_file.c)
target_link_libraries(eaf_play PRIVATE eaf_native_output)
if(EAF_ENABLE_SENDPIN)
  add_executable(eaf_sendspin_probe platform/native_linux/sendspin_probe.c)
  target_link_libraries(eaf_sendspin_probe PRIVATE eaf)
endif()
include(CTest)

  endif()

    add_executable(test_transports tests/test_transports.c)

    set_tests_properties(transports PROPERTIES TIMEOUT 30)

    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE eaf)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)

  if(EAF_ENABLE_SENDPIN)
    add_executable(test_sendspin_sync tests/test_sendspin_sync.c)
    target_link_libraries(test_sendspin_sync PRIVATE eaf)
    add_test(NAME sendspin_sync COMMAND test_sendspin_sync)
    set_tests_properties(sendspin_sync PROPERTIES TIMEOUT 30)

    add_test(NAME sendspin_protocol COMMAND test_sendspin_protocol)

    add_executable(test_sendspin_client tests/test_sendspin_client.c)
    target_link_libraries(test_sendspin_client PRIVATE eaf)
    target_link_options(test_sendspin_client PRIVATE -Wl,--wrap=hal_tcp_connect
      -Wl,--wrap=hal_tcp_send -Wl,--wrap=hal_tcp_recv -Wl,--wrap=hal_tcp_close
      -Wl,--wrap=hal_monotonic_time_us)
    add_test(NAME sendspin_client COMMAND test_sendspin_client)
    set_tests_properties(sendspin_client PROPERTIES TIMEOUT 30)
    add_executable(test_sendspin_player tests/test_sendspin_player.c)
    target_link_libraries(test_sendspin_player PRIVATE eaf)
    target_link_options(test_sendspin_player PRIVATE -Wl,--wrap=hal_monotonic_time_us)
    add_test(NAME sendspin_player COMMAND test_sendspin_player)
    set_tests_properties(sendspin_player PROPERTIES TIMEOUT 30)
  endif()
  target_link_options(test_hal_os PRIVATE -Wl,--wrap=sem_clockwait -Wl,--wrap=pthread_create)
  target_link_options(test_graph PRIVATE
    -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
  add_test(NAME native_demo COMMAND eaf_demo)
  set_tests_properties(native_demo PROPERTIES TIMEOUT 10)
