cmake_minimum_required(VERSION 3.20)
project(eaf VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(EAF_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(EAF_THREAD_SANITIZE "Thread sanitizer (separate build)" OFF)
option(EAF_ENABLE_LMS "Build SlimProto protocol support" ON)
option(EAF_ENABLE_BT_INGRESS "Build bounded SBC packet ingress" ON)
if(EAF_SANITIZE AND EAF_THREAD_SANITIZE)
  message(FATAL_ERROR "Select only one sanitizer mode")
endif()
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Werror -Wconversion -Wstrict-prototypes -Wshadow -Wdouble-promotion)
if(EAF_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(EAF_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(eaf STATIC core/eaf_types.c core/eaf_reservoir.c core/eaf_graph.c core/eaf_control.c
  core/eaf_dsp.c hal/common/hal_atomic_c11.c hal/linux/hal_os_linux.c
  hal/linux/hal_file_linux.c hal/common/sink_null.c
  apps/decoders/dec_wav.c apps/player/eaf_player.c)
target_include_directories(eaf PUBLIC include)
target_link_libraries(eaf PUBLIC Threads::Threads m)
if(EAF_ENABLE_LMS)
  target_sources(eaf PRIVATE apps/lms/lms_protocol.c apps/lms/lms_client.c hal/linux/hal_tcp_linux.c)
endif()
if(EAF_ENABLE_BT_INGRESS)
  target_sources(eaf PRIVATE apps/bt/bt_ingress.c apps/bt/bt_decoder.c)
endif()
include(cmake/libsbc.cmake)
add_executable(eaf_demo platform/native_linux/board_config.c)
target_link_libraries(eaf_demo PRIVATE eaf)
add_executable(eaf_play platform/native_linux/play_file.c)
target_link_libraries(eaf_play PRIVATE eaf)
include(CTest)
if(BUILD_TESTING)
  if(EAF_LIBSBC_ROOT AND EAF_ENABLE_BT_INGRESS)
    add_executable(test_sbc tests/test_sbc.c)
    target_link_libraries(test_sbc PRIVATE eaf)
    add_test(NAME sbc COMMAND test_sbc ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/tone.sbc)
  endif()
  if(EAF_ENABLE_LMS AND EAF_ENABLE_BT_INGRESS)
    add_executable(test_transports tests/test_transports.c)
    target_link_libraries(test_transports PRIVATE eaf)
    add_test(NAME transports COMMAND test_transports)
    set_tests_properties(transports PROPERTIES TIMEOUT 30)
  endif()
  foreach(name reservoir dsp graph spsc wav control player)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE eaf)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endforeach()
  target_link_options(test_graph PRIVATE
    -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
  add_test(NAME native_demo COMMAND eaf_demo)
  set_tests_properties(native_demo PROPERTIES TIMEOUT 10)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    if(EAF_ENABLE_LMS)
      add_executable(test_lms_peer tests/test_lms_peer.c)
      target_link_libraries(test_lms_peer PRIVATE eaf)
      add_test(NAME lms_peer COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lms_peer.py $<TARGET_FILE:test_lms_peer>)
      set_tests_properties(lms_peer PROPERTIES TIMEOUT 60)
    endif()
    add_test(NAME file_playback COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_play_file.py $<TARGET_FILE:eaf_play>)
    set_tests_properties(file_playback PROPERTIES TIMEOUT 30)
  endif()
endif()
