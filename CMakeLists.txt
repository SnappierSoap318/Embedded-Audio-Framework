cmake_minimum_required(VERSION 3.20)
project(eaf VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(EAF_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(EAF_THREAD_SANITIZE "Thread sanitizer (separate build)" OFF)
option(EAF_ENABLE_LMS "Build SlimProto protocol support" ON)
option(EAF_ENABLE_BT_INGRESS "Build bounded SBC packet ingress" ON)
if(EAF_SANITIZE AND EAF_THREAD_SANITIZE)
  message(FATAL_ERROR "Select only one sanitizer mode")
endif()
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Werror -Wconversion -Wstrict-prototypes -Wshadow -Wdouble-promotion)
if(EAF_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(EAF_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(eaf STATIC core/eaf_types.c core/eaf_reservoir.c core/eaf_graph.c core/eaf_dsp.c hal/common/hal_atomic_c11.c hal/linux/hal_os_linux.c hal/common/sink_null.c)
target_include_directories(eaf PUBLIC include)
target_link_libraries(eaf PUBLIC Threads::Threads m)
add_executable(eaf_demo platform/native_linux/board_config.c)
target_link_libraries(eaf_demo PRIVATE eaf)
include(CTest)
if(BUILD_TESTING)
  foreach(name reservoir dsp graph spsc)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE eaf)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endforeach()
  target_link_options(test_graph PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
  add_test(NAME native_demo COMMAND eaf_demo)
endif()
