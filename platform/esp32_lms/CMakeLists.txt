cmake_minimum_required(VERSION 3.20.0)
list(APPEND ZEPHYR_EXTRA_MODULES ${CMAKE_CURRENT_LIST_DIR}/../..)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(eaf_esp32_lms)
# Keep the optional silent backend within the WROOM budget; match all TUs.
zephyr_compile_definitions(EAF_NULL_MAX_FRAMES=128u)
set(EAF_PRIVATE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/credentials.local.h" CACHE FILEPATH "Local Wi-Fi credentials header")
if(EXISTS "${EAF_PRIVATE_CONFIG}")
  configure_file("${EAF_PRIVATE_CONFIG}" private/credentials.h COPYONLY)
else()
  configure_file(credentials.example.h private/credentials.h COPYONLY)
  message(STATUS "No local Wi-Fi credentials: image builds but will not connect")
endif()
target_include_directories(app PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/private)
target_sources(app PRIVATE src/main.c src/output.c src/credentials.c)
target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output/board_output.c)
target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/log_buffer.c
  ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/diagnostics.c)
target_include_directories(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output
  ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common)
if(CONFIG_EAF_BOARD_I2S)
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output/sink_i2s.c)
else()
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output/sink_null.c)
endif()
if(CONFIG_EAF_BOARD_USE_PSRAM)
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output/storage_psram.c)
else()
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_output/storage_internal.c)
endif()
if(CONFIG_SCHED_CPU_MASK)
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/cpu_pin_zephyr.c)
else()
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/cpu_pin_stub.c)
endif()
if(CONFIG_WIFI_ESP32)
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/wifi_power_esp32.c)
else()
  target_sources(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../esp32_common/wifi_power_stub.c)
endif()

target_include_directories(app SYSTEM PRIVATE ${ZEPHYR_BASE}/include)
target_compile_options(app PRIVATE -Wall -Wextra -Werror -Wconversion -Wstrict-prototypes -Wshadow -Wdouble-promotion)
